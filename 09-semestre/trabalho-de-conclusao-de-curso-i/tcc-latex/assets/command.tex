% Lista de figuras ========================================

% Renomear título da lista de figuras
\renewcommand{\listfigurename}{Lista de Figuras}

% Sem indentação da lista
\setlength{\cftfigureindent}{0pt} 

% Personalizar o formato (Figura 1)
\renewcommand{\cftfigurepresnum}{Figura\ } 

% Ajustar pontilhado (deixar sem nada)
\renewcommand{\cftdotsep}{\cftnodots} 

% Ajustar espaçamento entre número e texto
\setlength{\cftfigurenumwidth}{3cm} 

% Espaçamento entre linhas da lista
\setlength{\cftbeforefigureskip}{6pt} 

% Centralizar o título da Lista de Figuras
\renewcommand{\printloftitle}[1]{\centering\bfseries\Large #1}


% Lista de tabelas ========================================

% Renomear título da lista de tabelas
\renewcommand{\listtablename}{Lista de Tabelas}

% Sem indentação da lista
\setlength{\cfttableindent}{0pt} 

% Personalizar o formato (Tabela 1)
\renewcommand{\cfttablepresnum}{Tabela\ } 

% Ajustar espaçamento entre número e texto
\setlength{\cfttablenumwidth}{3cm} 

% Espaçamento entre linhas da lista
\setlength{\cftbeforetableskip}{6pt} 

% Centralizar o título da Lista de Tabelas
\renewcommand{\printlottitle}[1]{\centering\bfseries\Large #1}


% Sumário ========================================

% Renomear sumário
\renewcommand{\contentsname}{Sumário}

% Remove todos os recuos do Sumário
\setlength{\cftsectionindent}{0pt}
\setlength{\cftsubsectionindent}{0pt}
\setlength{\cftsubsubsectionindent}{0pt}

% Alinhamento perfeito dos números (ajustados para evitar sobreposição)
\setlength{\cftsectionnumwidth}{1cm}
\setlength{\cftsubsectionnumwidth}{1.2cm}
\setlength{\cftsubsubsectionnumwidth}{1.5cm}

% Controla a profundidade da numeração
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

% Centralizar o título do Sumário
\renewcommand{\printtoctitle}[1]{\centering\bfseries\Large #1}
