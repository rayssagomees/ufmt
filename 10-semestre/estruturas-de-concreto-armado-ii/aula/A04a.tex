            \item Momento de projeto
            
            $\gamma_c = 1,4$ \cite[tabela 12.1]{nbr6118} \newline
            
            $M_{d,x} = \gamma_c M_x = 1,4 \times 2,43 = 340,2 \ kN cm/m$
            
            $M_{d,y} = \gamma_c M_y = 1,4 \times 0,76 = 106,4 \ kN cm/m$ \newline

            \item Altura útil
            
            CAA-I = 20 mm  (laje) \cite[tabela 7.2]{nbr6118}
            
            $d = h - c -\phi_e - \frac{\phi_\ell}{2} = 10 \cdot 10^{-2} + \Big(-20 - 10 - \frac{10}{2}\Big)\cdot 10^{-3} = 6,5 \ cm$ \newline

            \item Taxa mínima de armadura
            
            $\rho_s = \frac{A_s}{A_c} \to A_c = bh$ \cite[tabela 19.1]{nbr6118}
            
            $\rho_{min} = 0,179\%$ \cite[tabela 17.3]{nbr6118} \newline
            
            $\rho_s \geq 0,67 \rho_{min} \text{\cite[tabela 19.1]{nbr6118}} \newline A_{s,min} \geq 0,67 \ \frac{0,179 \times 100 \times 10}{100} \newline A_{s,min} \geq 1,2 \ cm^2$ \newline
            
            \item Área de aço
            
            $K_{c,x} = \frac{b \ d^2}{M_d} = \frac{100 \times 6,5^2}{340,2} = 12,42 \ cm^2/kN$
            
            $K_{c,y} = \frac{b \ d^2}{M_d} = \frac{100 \times 6,5^2}{106,4} = 39,71 \ cm^2/kN$ \newline

            $\beta_{x,x} = 0,06 \leq 0,45 \to$ armadura simples, domínio 2 \cite[tabela 1.1a]{tabelaskcks}

            $\beta_{x,y} = 0,02 \leq 0,45 \to$ armadura simples, domínio 2 \cite[tabela 1.1a]{tabelaskcks} \newline

            \begin{observacao}
                $\begin{cases}
                    A_s \geq A_{s,min} \to A_s \\
                    A_s < A_{s,min} \to A_{s,min}
                \end{cases}$
            \end{observacao}

            $K_{s,x} = \frac{A_s \ d}{M_d} \longleftrightarrow A_s = \frac{K_s \ Md}{d} = \frac{0,024 \times 340,2}{6,5} = 1,26 \ cm^2/m$

            $K_{s,y} = \frac{A_s \ d}{M_d} \longleftrightarrow A_s = \frac{K_s \ Md}{d} = \frac{0,023 \times 106,4}{6,5} = 0,38\ cm^2/m$ \newline

            \item Determinação da bitola e do espaçamento
            
            $\phi_\ell \leq \frac{h}{8} = \frac{10}{8} = 1,25 \ cm = 12,5 \ mm$ \cite[item 20.1]{nbr6118} \newline

            \cite[item 20.1]{nbr6118}

            \begin{multicols}{2}
                $S_x \leq \begin{cases}
                    2h = 2 \times 10 = 20 \ cm \\
                    20 \ cm
                \end{cases}$

                $S_y \leq 33 \ cm$
            \end{multicols}

            $A_{\phi,x} = \pi\Big(\frac{D}{2}\Big)^2 = \pi \Big(\frac{5\cdot 10^{-3}}{2}\Big)^2 = 0,196 \ cm^2$ 

            \begin{observacao}
                O espaçamento deve ser arredondado para baixo.
            \end{observacao}
            
            $S_{\phi,x} = \frac{A_{\phi}}{A_s}\ 100 = \frac{0,196}{1,26} \ 100 = 15,58 \approx 15 \ cm$

            $\phi_{\ell,x} = 5 \ mm$ c/ $15 \ cm$ \newline

            %$A_{\phi,y} = \pi\Big(\frac{D}{2}\Big)^2 = \pi \Big(\frac{5\cdot 10^{-3}}{2}\Big)^2 = 0,196 \ cm^2$ 
            
            $S_{\phi,y} = \frac{A_{\phi}}{A_s}\ 100 = \frac{0,196}{1,2} \ 100 = 16,36 \approx 16 \ cm$ 

            $\phi_{\ell,y} = 5 \ mm$ c/ $16 \ cm$\newline
        \end{enumerate}
