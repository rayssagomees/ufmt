                $f_{ctk,inf} = 0,7 \ f_{ct,m}$ \cite[item 8.2.5]{nbr6118}

                $f_{ct,m} = 0,3 f_{ck}^{2/3}$ \cite[item 8.2.5]{nbr6118}
